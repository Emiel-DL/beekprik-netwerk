  snap_and_attribute <- function(points, network, grp_col, tol, vhas_col_pts, vhas_col_net) {

    points$node_id <- NA

    if (!is.null(grp_col)) points$match_group_val <- NA



    # 1. VHAS MATCHING

    has_vhas_config <- !is.null(vhas_col_pts) && !is.null(vhas_col_net)

    can_use_vhas <- has_vhas_config && (vhas_col_pts %in% names(points)) && (vhas_col_net %in% names(network))



    if (can_use_vhas) {

      pts_with_vhas_idx <- which(!is.na(points[[vhas_col_pts]]))

      if (length(pts_with_vhas_idx) > 0) {

        unique_codes <- unique(points[[vhas_col_pts]][pts_with_vhas_idx])

        for (code in unique_codes) {

          net_idx <- which(network[[vhas_col_net]] == code)

          if (length(net_idx) > 0) {

            p_subset_idx <- which(points[[vhas_col_pts]] == code)

            rel_nearest <- st_nearest_feature(points[p_subset_idx,], network[net_idx,])

            abs_nearest <- net_idx[rel_nearest]

            points$node_id[p_subset_idx] <- as.character(network$from_node[abs_nearest])

            if (!is.null(grp_col)) points$match_group_val[p_subset_idx] <- network[[grp_col]][abs_nearest]

          }

        }

      }

    }



    # 2. GEOMETRIC FALLBACK

    missing_idx <- which(is.na(points$node_id))

    if (length(missing_idx) > 0) {

      subset_pts <- points[missing_idx, ]

      nearest_idx <- st_nearest_feature(subset_pts, network)

      dists <- st_distance(subset_pts, network[nearest_idx, ], by_element = TRUE)

      dists_num <- as.numeric(dists)

      dists_num[is.na(dists_num)] <- Inf



      valid_snaps <- dists_num <= tol

      valid_global_idx <- missing_idx[valid_snaps]

      valid_net_idx <- nearest_idx[valid_snaps]



      if (length(valid_global_idx) > 0) {

        points$node_id[valid_global_idx] <- as.character(network$from_node[valid_net_idx])

        if (!is.null(grp_col)) points$match_group_val[valid_global_idx] <- network[[grp_col]][valid_net_idx]

      }



      n_dropped <- length(missing_idx) - length(valid_global_idx)

      if (n_dropped > 0) message(paste("      ->", n_dropped, "points dropped (too far/no VHAS match)."))

    }

    return(points)

  }



  biota_sf <- snap_and_attribute(biota_sf, net_sf, grouping_col, snap_tolerance, vhas_col_biota, vhas_col_network)

        upstream_nodes <- names(subcomponent(g, v = b_node, mode = "in"))

        # Bereken graafafstand (alleen zinvol voor upstream, maar we runnen voor subset)

        d_matrix <- distances(g, v = subset_quality$node_id, to = b_node, mode = "out")

        # B. Euclidische Afstand (Vogelvlucht)

        # Dit doen we via de index die we eerder hebben opgeslagen

        candidates <- subset_quality %>%

        # STAP 4: Bepaal de 'definitive' afstand voor sortering

        # Als het upstream is -> Graph dist

        # Als het downstream is (graph=Inf) -> Euclid dist

        # Als het beide is (kortbij upstream) -> Graph dist (consistentie)

                                            candidates$graph_dist,

          if (length(discharge_nodes) > 0) {

            }

        }

    results[[i]] <- match

  }



  message("3/5 Merging results...")

  matched_quality_df <- do.call(rbind, results)

  cols_to_rename <- setdiff(names(matched_quality_df), "biota_match_id")

  names(matched_quality_df)[names(matched_quality_df) %in% cols_to_rename] <- paste0("qual_", cols_to_rename)

  final_sf <- left_join(biota_sf, matched_quality_df, by = c("tmp_join_id" = "biota_match_id"))

  if("strahler_val" %in% names(final_sf)) final_sf$strahler_val <- NULL



  return(final_sf)

In plaats van koppelen van MI met quality punten geeft ik een laag met observaties van beekprik over verschillende jaren. nu wil ik op basis van die observaties voor bepaalde tijdsvensters die te specifieren zijn alle segmenten hebben die op x km van een observatie liggen om zo te komen tot areaalclusters (multilinestring) van beekprik.

Gemini zei
Dit is een interessante verschuiving in de logica. In plaats van een point-to-point koppeling (Biota naar Quality), gaan we nu naar een point-to-network selectie. Je wilt een "buffer" op het netwerk genereren rondom je observaties.

Omdat je met 164.000 segmenten werkt, is efficiëntie cruciaal. Hieronder vind je de aangepaste functie get_beekprik_clusters.

De aanpak
Tijdsfilter: We filteren de observaties (beekprik) eerst op het gewenste jaar/venster.

Network Snapping: We bepalen op welke node in het netwerk de beekprik ligt.

Graph Traversal: We gebruiken igraph::distances om in één klap alle nodes te vinden die binnen X km liggen (zowel upstream als downstream, aangezien een cluster zich in beide richtingen verspreidt).

Segment Selectie: We halen alle lijnsegmenten op die verbonden zijn met deze "bereikbare" nodes.

Unificatie: We voegen de segmenten samen tot MULTILINESTRING per cluster of jaar.

  snap_and_attribute <- function(points, network, grp_col, tol, vhas_col_pts, vhas_col_net) {

    points$node_id <- NA

    if (!is.null(grp_col)) points$match_group_val <- NA



    # 1. VHAS MATCHING

    has_vhas_config <- !is.null(vhas_col_pts) && !is.null(vhas_col_net)

    can_use_vhas <- has_vhas_config && (vhas_col_pts %in% names(points)) && (vhas_col_net %in% names(network))



    if (can_use_vhas) {

      pts_with_vhas_idx <- which(!is.na(points[[vhas_col_pts]]))

      if (length(pts_with_vhas_idx) > 0) {

        unique_codes <- unique(points[[vhas_col_pts]][pts_with_vhas_idx])

        for (code in unique_codes) {

          net_idx <- which(network[[vhas_col_net]] == code)

          if (length(net_idx) > 0) {

            p_subset_idx <- which(points[[vhas_col_pts]] == code)

            rel_nearest <- st_nearest_feature(points[p_subset_idx,], network[net_idx,])

            abs_nearest <- net_idx[rel_nearest]

            points$node_id[p_subset_idx] <- as.character(network$from_node[abs_nearest])

            if (!is.null(grp_col)) points$match_group_val[p_subset_idx] <- network[[grp_col]][abs_nearest]

          }

        }

      }

    }



    # 2. GEOMETRIC FALLBACK

    missing_idx <- which(is.na(points$node_id))

    if (length(missing_idx) > 0) {

      subset_pts <- points[missing_idx, ]

      nearest_idx <- st_nearest_feature(subset_pts, network)

      dists <- st_distance(subset_pts, network[nearest_idx, ], by_element = TRUE)

      dists_num <- as.numeric(dists)

      dists_num[is.na(dists_num)] <- Inf



      valid_snaps <- dists_num <= tol

      valid_global_idx <- missing_idx[valid_snaps]

      valid_net_idx <- nearest_idx[valid_snaps]



      if (length(valid_global_idx) > 0) {

        points$node_id[valid_global_idx] <- as.character(network$from_node[valid_net_idx])

        if (!is.null(grp_col)) points$match_group_val[valid_global_idx] <- network[[grp_col]][valid_net_idx]

      }



      n_dropped <- length(missing_idx) - length(valid_global_idx)

      if (n_dropped > 0) message(paste("      ->", n_dropped, "points dropped (too far/no VHAS match)."))

    }

    return(points)

  }



  biota_sf <- snap_and_attribute(biota_sf, net_sf, grouping_col, snap_tolerance, vhas_col_biota, vhas_col_network)

  quality_sf <- snap_and_attribute(quality_sf, net_sf, grouping_col, snap_tolerance, vhas_col_quality, vhas_col_network)

      dists <- st_distance(pts, str_layer[idx, ], by_element = TRUE)

      vals <- rep(NA, nrow(pts))

      valid <- dists_num <= tol

      vals[valid] <- str_layer[[str_col]][idx[valid]]

      return(vals)

    }

  }



  quality_df <- st_drop_geometry(quality_sf)



  # --- Discharge Snapping ---

  discharge_nodes <- character(0)

  if (!is.null(discharge_sf)) {

    message("    -> Snapping discharge points...")

    idx_dis <- st_nearest_feature(discharge_sf, net_sf)

    dists_dis <- st_distance(discharge_sf, net_sf[idx_dis, ], by_element = TRUE)

    dists_num <- as.numeric(dists_dis)

    dists_num[is.na(dists_num)] <- Inf

    valid_dis <- dists_num <= snap_tolerance

    if (sum(valid_dis) > 0) {

    }

  }

  results <- list()

  pb <- progress_bar$new(format = "  [:bar] :percent eta: :eta", total = nrow(biota_sf), clear = FALSE, width = 60)



  for (i in seq_len(nrow(biota_sf))) {

    pb$tick()

    b_pt <- biota_sf[i, ]

    b_node <- b_pt$node_id

    b_id   <- b_pt$tmp_join_id

    match <- NA

    n_matches_found <- 0



    if (is.na(b_node)) {

      match <- quality_df[1, ]

      match[] <- NA; match$river_dist_m <- NA

    } else {

      b_date <- b_pt[[col_date_biota]]



      # STAP 1: Pas eerst alle niet-ruimtelijke filters toe (Datum, Groep, Strahler)

      subset_quality <- quality_df %>% filter(!is.na(node_id))



      if (!is.null(grouping_col)) {

        b_group_val <- b_pt$match_group_val

        if (!is.na(b_group_val)) {

          subset_quality <- subset_quality %>% filter(match_group_val == b_group_val)

        } else {

          subset_quality <- subset_quality[0, ]

        }

      }



      if (use_strahler && nrow(subset_quality) > 0) {

        b_strahler <- b_pt$strahler_val

        if (!is.na(b_strahler)) {

          subset_quality <- subset_quality %>% filter(!is.na(strahler_val), abs(strahler_val - b_strahler) <= 1)

        } else {

          subset_quality <- subset_quality[0, ]

        }

      }



      # Filter op tijd

      if (nrow(subset_quality) > 0) {

        subset_quality <- subset_quality %>%

          mutate(

            temp_diff_days = as.numeric(difftime(b_date, .[[col_date_quality]], units = "days")),

            abs_time_diff  = abs(temp_diff_days)

          ) %>%

          filter(temp_diff_days >= -days_after & temp_diff_days <= days_before)

      }



      if (nrow(subset_quality) > 0) {



        # STAP 2: Bereken afstanden (Graaf EN Euclidisch)



        # A. Graaf Afstand (alleen upstream werkt hier, downstream geeft Inf)

        upstream_nodes <- names(subcomponent(g, v = b_node, mode = "in"))

        subset_quality$is_upstream <- subset_quality$node_id %in% upstream_nodes



        # Bereken graafafstand (alleen zinvol voor upstream, maar we runnen voor subset)

        d_matrix <- distances(g, v = subset_quality$node_id, to = b_node, mode = "out")

        subset_quality$graph_dist <- as.numeric(d_matrix)



        # B. Euclidische Afstand (Vogelvlucht)

        # We moeten de geometry ophalen uit de originele quality_sf

        # Dit doen we via de index die we eerder hebben opgeslagen

        target_geoms <- quality_sf[subset_quality$orig_index_geo, ]

        euclid_dists <- st_distance(b_pt, target_geoms)

        subset_quality$euclid_dist <- as.numeric(euclid_dists)



        # STAP 3: De 'OR' Filter

        # Kandidaat is geldig als:

        # (Is Upstream EN GraphDist <= 5000) OF (EuclidDist <= 200)



        candidates <- subset_quality %>%

          filter(

            (is_upstream & graph_dist <= max_dist_m) |

              (euclid_dist <= max_downstream_m)

          )



        # STAP 4: Bepaal de 'definitive' afstand voor sortering

        # Als het upstream is -> Graph dist

        # Als het downstream is (graph=Inf) -> Euclid dist

        # Als het beide is (kortbij upstream) -> Graph dist (consistentie)

        if (nrow(candidates) > 0) {

          candidates$river_dist_m <- ifelse(is.finite(candidates$graph_dist),

                                            candidates$graph_dist,

                                            candidates$euclid_dist)



          # Topology check (Lozingen) - enkel relevant voor upstream paden!

          # Als een punt downstream ligt, heeft een lozing stroomopwaarts (tussen biota en qual) geen invloed op qual.

          if (length(discharge_nodes) > 0) {

            # We checken alleen de kandidaten die 'upstream' zijn volgens de graaf

            to_check_idx <- which(candidates$is_upstream & candidates$river_dist_m > 0)



            if(length(to_check_idx) > 0) {

              keep_flags <- rep(TRUE, nrow(candidates))



              checks <- sapply(candidates$node_id[to_check_idx], function(cand_node) {

                if (cand_node == b_node) return(TRUE)

                path_res <- shortest_paths(g, from = cand_node, to = b_node, mode = "out")

                intermediate_nodes <- setdiff(names(path_res$vpath[[1]]), c(cand_node, b_node))

                if (any(intermediate_nodes %in% discharge_nodes)) return(FALSE) else return(TRUE)

              })



              keep_flags[to_check_idx] <- checks

              candidates <- candidates[keep_flags, ]

            }

          }



          candidates_within <- candidates # Naamgeving consistent houden



          if (nrow(candidates_within) > 0) {

            # ... (Selectie logica blijft hetzelfde) ...

            if (selection_mode == "closest_distance") {

              match <- candidates_within[which.min(candidates_within$river_dist_m), ]; n_matches_found <- 1

            } else if (selection_mode == "closest_time") {

              match <- candidates_within[which.min(candidates_within$abs_time_diff), ]; n_matches_found <- 1

            } else if (selection_mode == "all") {

              match <- candidates_within; n_matches_found <- nrow(candidates_within)

            } else if (selection_mode == "aggregate") {

              n_matches_found <- nrow(candidates_within)

              if (!is.null(aggr_cols)) { cols_to_agg <- aggr_cols } else {

                tech_cols <- c("node_id", "river_dist_m", "temp_diff_days", "abs_time_diff", "strahler_val", "match_group_val", "orig_index_geo", "graph_dist", "euclid_dist", "is_upstream")

                num_cols <- names(select_if(candidates_within, is.numeric))

                cols_to_agg <- setdiff(num_cols, tech_cols)

              }

              agg_fun <- switch(aggr_method, "mean"=function(x) mean(x, na.rm=T), "min"=function(x) min(x, na.rm=T), "max"=function(x) max(x, na.rm=T))



              if (length(cols_to_agg) > 0) {

                agg_values <- candidates_within %>% summarise(across(all_of(cols_to_agg), agg_fun))

                match <- candidates_within[1, ]

                match[, cols_to_agg] <- agg_values

              } else { match <- candidates_within[1, ] }

              match$river_dist_m <- mean(candidates_within$river_dist_m)

            }

          }

        }

      }

      if (!is.data.frame(match)) { match <- quality_df[1, ]; match[] <- NA; match$river_dist_m <- NA }

    }

    match$biota_match_id <- b_id

    match$n_matches <- n_matches_found

    # Cleanup extra kolommen

    cols_to_remove <- c("temp_diff_days", "abs_time_diff", "match_group_val", "strahler_val", "orig_index_geo", "graph_dist", "euclid_dist", "is_upstream")

    match[, base::intersect(names(match), cols_to_remove)] <- NULL

    results[[i]] <- match

  }



  message("3/5 Merging results...")

  matched_quality_df <- do.call(rbind, results)

  cols_to_rename <- setdiff(names(matched_quality_df), "biota_match_id")

  names(matched_quality_df)[names(matched_quality_df) %in% cols_to_rename] <- paste0("qual_", cols_to_rename)

  final_sf <- left_join(biota_sf, matched_quality_df, by = c("tmp_join_id" = "biota_match_id"))



  final_sf$tmp_join_id <- NULL

  if("match_group_val" %in% names(final_sf)) final_sf$match_group_val <- NULL

  if("strahler_val" %in% names(final_sf)) final_sf$strahler_val <- NULL



  return(final_sf)

}

In plaats van koppelen van MI met quality punten geeft ik een laag met observaties van beekprik over verschillende jaren. nu wil ik op basis van die observaties voor bepaalde tijdsvensters die te specifieren zijn alle segmenten hebben die op x km van een observatie liggen om zo te komen tot areaalclusters (multilinestring) van beekprik.

Gemini zei
Dit is een heel interessante verschuiving in je logica! In plaats van een point-to-point koppeling (Biota → Quality), wil je nu een point-to-network analyse doen. Je wilt feitelijk een stroomopwaartse en stroomafwaartse "buffer" (of isochroon) over je waterloopnetwerk leggen rondom je observaties om zo tot aaneengesloten habitatclusters te komen.

Omdat je met ruim 164.000 segmenten werkt, is efficiëntie hier de sleutel. In plaats van voor elke observatie apart het netwerk af te speuren en dan alles samen te voegen, kunnen we de igraph afstandsberekening optimaliseren door in één keer de hele matrix te berekenen en de verbonden componenten (clusters) op te sporen.

Hier is de aangepaste R functie get_beekprik_clusters. Ik heb de functie zo geschreven dat hij niet alleen de segmenten selecteert, maar ook ruimtelijk gescheiden clusters identificeert en deze netjes samenvoegt tot afzonderlijke MULTILINESTRING features.

get_beekprik_clusters <- function(observations_sf, 
                                  network_list, 
                                  target_years = NULL, 
                                  max_dist_m = 5000, 
                                  snap_tolerance = 50,
                                  time_col = "jaar") {
  
  message("1/5 Validatie en filteren van de data...")
  g <- network_list$graph
  net_sf <- network_list$network_sf
  
  # 1. Tijdsfilter toepassen (indien opgegeven)
  if (!is.null(target_years)) {
    obs_filtered <- observations_sf %>% filter(.data[[time_col]] %in% target_years)
  } else {
    obs_filtered <- observations_sf
  }
  
  if (nrow(obs_filtered) == 0) {
    warning("Geen observaties gevonden voor de opgegeven criteria.")
    return(NULL)
  }
  
  message(paste(" ->", nrow(obs_filtered), "observaties geselecteerd voor de analyse."))
  
  # 2. Snappen van observaties aan het netwerk
  message("2/5 Observaties snappen aan de graaf...")
  nearest_idx <- st_nearest_feature(obs_filtered, net_sf)
  dists <- st_distance(obs_filtered, net_sf[nearest_idx, ], by_element = TRUE)
  
  valid_snaps <- as.numeric(dists) <= snap_tolerance
  if (sum(valid_snaps) == 0) {
    warning("Geen enkele observatie lag binnen de snap_tolerance van het netwerk.")
    return(NULL)
  }
  
  # Haal de unieke startknooppunten op
  # We gebruiken de from_node van het dichtstbijzijnde segment
  snapped_nodes <- unique(as.character(net_sf$from_node[nearest_idx[valid_snaps]]))
  message(paste(" ->", length(snapped_nodes), "unieke netwerk-nodes geïdentificeerd als startpunten."))
  
  # 3. Bereken netwerkafstanden
  message(paste("3/5 Netwerk verkennen (max afstand:", max_dist_m, "m)..."))
  # mode = "all" is cruciaal: vissen kunnen stroomopwaarts én stroomafwaarts zwemmen
  # distances() geeft een matrix: Rijen = snapped_nodes, Kolommen = alle nodes in de graaf
  dist_matrix <- distances(g, v = snapped_nodes, mode = "all")
  
  # Voor elke node in de graaf zoeken we de minimale afstand tot EENDER WELKE observatie
  # apply met MARGIN=2 betekent: voer uit per kolom
  min_dists <- apply(dist_matrix, 2, min)
  
  # Selecteer alle nodes die binnen de drempelwaarde vallen
  valid_nodes <- colnames(dist_matrix)[is.finite(min_dists) & min_dists <= max_dist_m]
  
  # 4. Losse clusters identificeren via een sub-graaf
  message("4/5 Ruimtelijk gescheiden clusters identificeren...")
  # We maken een tijdelijke graaf die enkel bestaat uit de bereikbare nodes
  sub_g <- induced_subgraph(g, valid_nodes)
  
  # Bepaal de verbonden componenten (dit zijn onze losse areaalclusters!)
  comps <- components(sub_g)
  
  # Maak een tabelletje met Node ID en bijbehorend Cluster ID
  node_clusters <- data.frame(
    node_id = names(comps$membership),
    cluster_id = as.character(comps$membership)
  )
  
  # 5. Segmenten ophalen en samenvoegen tot MULTILINESTRING
  message("5/5 Geometrieën samenvoegen (Dissolve)...")
  
  # We koppelen de segmenten op basis van de from_node. 
  # (Omdat elk segment in de sub-graaf verweven is, is dit voldoende)
  cluster_segments <- net_sf %>%
    inner_join(node_clusters, by = c("from_node" = "node_id"))
  
  # Groepeer per cluster en smelt de geometrieën samen
  final_clusters_sf <- cluster_segments %>%
    group_by(cluster_id) %>%
    summarise(
      n_segmenten = n(),
      totale_lengte_km = round(sum(weight) / 1000, 2), # weight was lengte in meters
      geometry = st_union(geometry), # Smelt lijnen samen
      .groups = "drop"
    ) %>%
    # Zorg dat de output netjes één of meerdere MULTILINESTRINGS zijn
    st_cast("MULTILINESTRING") 
  
  message(paste("Klaar! Er zijn", nrow(final_clusters_sf), "aaneengesloten beekprik-clusters gegenereerd."))
  
  return(final_clusters_sf)
}

Gemini zei
Dit is een heel scherpe observatie! De topologie (het doorstroombare netwerk) verandert inderdaad door de tijd heen naarmate knelpunten worden opgelost of erbij komen.

Hoe pakken we dit aan?
De enige correcte manier om dit op te lossen, is door de analyse jaar per jaar te itereren. We maken de netwerkgraaf dynamisch:

We nemen een specifiek jaar (bv. 2020).

We filteren de waarnemingen van 2020.

We filteren de actieve knelpunten in 2020.

We 'knippen' de actieve knelpunten letterlijk uit de netwerkgraaf van dat jaar (igraph::delete_vertices). Hierdoor kan het algoritme onmogelijk stroomopwaarts of -afwaarts door dit punt rekenen.

We berekenen de clusters voor 2020 en slaan deze op.

We herhalen dit voor het volgende jaar (bv. 2021). Is het knelpunt daar opgelost? Dan filtert stap 3 hem niet meer mee, blijft de node in de graaf, en kan de beekprik er in 2021 wél doorheen "zwemmen" in de analyse.

Hier is de volledig aangepaste functie die dit per jaar afhandelt en de hybrid-snapping voor knelpunten integreert.

De Aangepaste R Functie
R
get_beekprik_clusters_tijdreeks <- function(observations_sf, 
                                            network_list, 
                                            knelpunten_sf = NULL,
                                            target_years = NULL, 
                                            max_dist_m = 5000, 
                                            snap_tolerance_obs = 50,
                                            snap_tolerance_knelpunt = 25,
                                            time_col = "jaar",
                                            vhas_col_knelpunt = "vhas",
                                            vhas_col_network = "VHAS") {
  
  # --- 1. Voorbereiding ---
  g_base <- network_list$graph
  net_sf <- network_list$network_sf
  
  if (!is.null(target_years)) {
    observations_sf <- observations_sf %>% filter(.data[[time_col]] %in% target_years)
  }
  
  if (nrow(observations_sf) == 0) {
    warning("Geen observaties gevonden voor de opgegeven criteria.")
    return(NULL)
  }
  
  unieke_jaren <- sort(unique(observations_sf[[time_col]]))
  message(paste("Analyse gestart voor", length(unieke_jaren), "verschillende jaren..."))
  
  # Lijst om resultaten per jaar in op te slaan
  all_clusters <- list()
  
  # --- 2. Iteratie per jaar ---
  for (yr in unieke_jaren) {
    message(paste0("\n--- Verwerken van jaar: ", yr, " ---"))
    
    # A. Data filteren voor huidig jaar
    obs_yr <- observations_sf %>% filter(.data[[time_col]] == yr)
    message(paste(" ->", nrow(obs_yr), "observaties."))
    
    # B. Observaties snappen
    idx_obs <- st_nearest_feature(obs_yr, net_sf)
    dist_obs <- as.numeric(st_distance(obs_yr, net_sf[idx_obs, ], by_element = TRUE))
    valid_obs <- dist_obs <= snap_tolerance_obs
    
    start_nodes <- unique(as.character(net_sf$from_node[idx_obs[valid_obs]]))
    if (length(start_nodes) == 0) {
      message(" -> Geen observaties succesvol gesnapt binnen de tolerantie. Overslaan.")
      next
    }
    
    # C. Knelpunten voor dit jaar bepalen en snappen
    barrier_nodes <- character(0)
    
    if (!is.null(knelpunten_sf)) {
      knel_yr <- knelpunten_sf %>% filter(.data[[time_col]] == yr)
      
      if (nrow(knel_yr) > 0) {
        snapped_barriers <- character(nrow(knel_yr))
        
        for (i in seq_len(nrow(knel_yr))) {
          k_pt <- knel_yr[i, ]
          k_vhas <- k_pt[[vhas_col_knelpunt]]
          match_found <- FALSE
          
          # Poging 1: Matchen via VHAS
          if (!is.na(k_vhas) && k_vhas != "") {
            net_idx <- which(net_sf[[vhas_col_network]] == k_vhas)
            if (length(net_idx) > 0) {
              rel_idx <- st_nearest_feature(k_pt, net_sf[net_idx, ])
              abs_idx <- net_idx[rel_idx]
              dist_k <- as.numeric(st_distance(k_pt, net_sf[abs_idx, ]))
              
              if (dist_k <= snap_tolerance_knelpunt) {
                snapped_barriers[i] <- as.character(net_sf$from_node[abs_idx])
                match_found <- TRUE
              }
            }
          }
          
          # Poging 2: Fallback puur geometrisch
          if (!match_found) {
            abs_idx <- st_nearest_feature(k_pt, net_sf)
            dist_k <- as.numeric(st_distance(k_pt, net_sf[abs_idx, ]))
            if (dist_k <= snap_tolerance_knelpunt) {
              snapped_barriers[i] <- as.character(net_sf$from_node[abs_idx])
            } else {
              snapped_barriers[i] <- NA
            }
          }
        }
        
        barrier_nodes <- unique(na.omit(snapped_barriers))
        message(paste(" ->", length(barrier_nodes), "actieve migratieknelpunten succesvol gesnapt."))
      } else {
        message(" -> Geen actieve migratieknelpunten in dit jaar.")
      }
    }
    
    # D. Graaf aanpassen voor het huidige jaar
    # Let op: als een observatie *exact* op een knelpunt valt, halen we die start_node eruit 
    # anders faalt het algoritme. Hij verspreidt dan niet.
    start_nodes <- setdiff(start_nodes, barrier_nodes)
    
    if (length(start_nodes) == 0) {
      message(" -> Alle startpunten vallen exact op een knelpunt en zijn geblokkeerd. Overslaan.")
      next
    }
    
    # Verwijder de knooppunten uit de graaf. Hierdoor is de verbinding daar definitief doorgeknipt.
    g_yr <- g_base
    if (length(barrier_nodes) > 0) {
      # igraph filtert op vertex names
      nodes_to_delete <- V(g_yr)[name %in% barrier_nodes]
      g_yr <- delete_vertices(g_yr, nodes_to_delete)
    }
    
    # E. Netwerkafstanden en clusters berekenen
    dist_matrix <- distances(g_yr, v = start_nodes, mode = "all")
    min_dists <- apply(dist_matrix, 2, min)
    valid_nodes <- colnames(dist_matrix)[is.finite(min_dists) & min_dists <= max_dist_m]
    
    # Componenten bepalen op basis van bereikbare gebieden
    sub_g <- induced_subgraph(g_yr, valid_nodes)
    comps <- components(sub_g)
    
    node_clusters <- data.frame(
      node_id = names(comps$membership),
      cluster_id_temp = as.character(comps$membership)
    )
    
    # F. Samenvoegen tot MULTILINESTRING en jaar toevoegen
    cluster_segments <- net_sf %>%
      inner_join(node_clusters, by = c("from_node" = "node_id"))
    
    clusters_sf_yr <- cluster_segments %>%
      group_by(cluster_id_temp) %>%
      summarise(
        jaar = yr,
        n_segmenten = n(),
        totale_lengte_km = round(sum(weight) / 1000, 2),
        geometry = st_union(geometry),
        .groups = "drop"
      ) %>%
      mutate(cluster_id = paste0("Y", yr, "_C", cluster_id_temp)) %>%
      select(cluster_id, jaar, n_segmenten, totale_lengte_km, geometry) %>%
      st_cast("MULTILINESTRING")
    
    all_clusters[[as.character(yr)]] <- clusters_sf_yr
    message(paste(" ->", nrow(clusters_sf_yr), "clusters gegenereerd voor", yr))
  }
  
  # --- 3. Resultaten bundelen ---
  if (length(all_clusters) > 0) {
    final_clusters_sf <- bind_rows(all_clusters)
    message(paste("\nKlaar! In totaal", nrow(final_clusters_sf), "clusters gegenereerd over alle jaren heen."))
    return(final_clusters_sf)
  } else {
    message("\nKlaar! Geen clusters kunnen genereren.")
    return(NULL)
  }
}
De logica uitgelegd
De VHAS/Geometrie Fallback (Stap 2C): Ik heb exact de logica toegepast die je in de vorige iteratie gebruikte. Hij zoekt eerst in de stroomgebieden of de vhas code van het knelpunt ergens te bespeuren valt binnen 25m. Vindt hij niks? Dan valt hij terug op een harde geometrische nabijheidszoektocht.

Tijds-dynamische Graaf (delete_vertices): Dit is de magic trick. Door per tijdsstap (yr) via igraph::delete_vertices de gesnapte knelpunten uit de graaf te slopen, verbreek je de verbinding daar. distances() zal vanaf de beekprik-locatie dat knelpunt bereiken, en dan domweg zien dat de weg stopt (afstand tot de volgende node wordt de facto overschreven met Inf).

Dynamische Cluster ID's: Omdat een cluster in 2019 anders kan zijn dan één in 2020 (bijvoorbeeld door een opgelost knelpunt), krijgt elke gegenereerde cluster een uniek ID op basis van het jaar, bv. Y2019_C1.

Ik heb de code hieronder flink geoptimaliseerd met twee grote trucs:

Vectorisatie van de knelpunten: In plaats van het hele netwerk rij per rij af te speuren voor elk knelpunt, doet hij de geometrische "fallback" nu voor álle knelpunten in één klap. Dat scheelt seconden tot minuten per jaar.

De "Super Node" truc (Graaf-theorie): In plaats van alle afstanden vanaf élke observatie afzonderlijk te berekenen, voeg ik tijdelijk een onzichtbaar centraal knooppunt toe dat met afstand 0 verbonden is aan al je observaties. Het netwerk hoeft nu maar 1 keer te rekenen vanaf dat knooppunt, in plaats van 50 keer. Dit haalt de rekentijd voor het netwerk drastisch naar beneden.

Hier is de geoptimaliseerde versie. Probeer deze eens, je zou een enorm verschil in snelheid moeten merken:

R
get_beekprik_clusters_tijdreeks_fast <- function(observations_sf, 
                                                 network_list, 
                                                 knelpunten_sf = NULL,
                                                 target_years = NULL, 
                                                 max_dist_m = 5000, 
                                                 snap_tolerance_obs = 50,
                                                 time_col = "jaar",
                                                 vhas_col_knelpunt = "vhas",
                                                 vhas_col_network = "VHAS") {
  
  # --- 1. Voorbereiding ---
  g_base <- network_list$graph
  net_sf <- network_list$network_sf
  
  if (!is.null(target_years)) {
    observations_sf <- observations_sf %>% filter(.data[[time_col]] %in% target_years)
  }
  
  if (nrow(observations_sf) == 0) return(NULL)
  
  unieke_jaren <- sort(unique(observations_sf[[time_col]]))
  message(sprintf("Analyse gestart voor %d jaren...", length(unieke_jaren)))
  
  all_clusters <- list()
  
  # --- 2. Iteratie per jaar ---
  for (yr in unieke_jaren) {
    message(paste0("\n--- Verwerken van jaar: ", yr, " ---"))
    
    # A. Data filteren voor huidig jaar
    obs_yr <- observations_sf %>% filter(.data[[time_col]] == yr)
    
    # B. Observaties snappen
    idx_obs <- st_nearest_feature(obs_yr, net_sf)
    dist_obs <- as.numeric(st_distance(obs_yr, net_sf[idx_obs, ], by_element = TRUE))
    valid_obs <- dist_obs <= snap_tolerance_obs
    
    start_nodes <- unique(as.character(net_sf$from_node[idx_obs[valid_obs]]))
    if (length(start_nodes) == 0) {
      message(" -> Geen observaties gesnapt binnen tolerantie. Overslaan.")
      next
    }
    
    # C. Knelpunten supersnel snappen (Vectorised)
    barrier_nodes <- character(0)
    if (!is.null(knelpunten_sf)) {
      knel_yr <- knelpunten_sf %>% filter(.data[[time_col]] == yr)
      
      if (nrow(knel_yr) > 0) {
        # Stap 1: Doe de geometrische fallback voor ALLES tegelijk in 1 bewerking (razendsnel)
        fallback_idx <- st_nearest_feature(knel_yr, net_sf)
        snapped_barriers <- as.character(net_sf$from_node[fallback_idx])
        
        # Stap 2: Overschrijf enkel degene met een geldige VHAS-match
        # (Dit is snel omdat we enkel zoeken binnen een piepkleine subset van het netwerk)
        for (i in seq_len(nrow(knel_yr))) {
          k_vhas <- knel_yr[[vhas_col_knelpunt]][i]
          if (!is.na(k_vhas) && k_vhas != "") {
            net_idx <- which(net_sf[[vhas_col_network]] == k_vhas)
            if (length(net_idx) > 0) {
              rel_idx <- st_nearest_feature(knel_yr[i, ], net_sf[net_idx, ])
              snapped_barriers[i] <- as.character(net_sf$from_node[net_idx[rel_idx]])
            }
          }
        }
        barrier_nodes <- unique(na.omit(snapped_barriers))
        message(paste(" ->", length(barrier_nodes), "knelpunten gesnapt."))
      }
    }
    
    # D. Graaf knippen
    start_nodes <- setdiff(start_nodes, barrier_nodes)
    if (length(start_nodes) == 0) {
      message(" -> Alle startpunten geblokkeerd. Overslaan.")
      next
    }
    
    g_yr <- g_base
    if (length(barrier_nodes) > 0) {
      nodes_to_delete <- V(g_yr)[name %in% barrier_nodes]
      g_yr <- delete_vertices(g_yr, nodes_to_delete)
    }
    
    # E. SUPER NODE TRICK voor snelle routeberekening
    # Voeg een dummy node toe en verbind hem met afstand 0 aan alle start_nodes
    g_yr <- add_vertices(g_yr, 1, name = "SUPER_START")
    edges_to_add <- as.vector(rbind("SUPER_START", start_nodes))
    g_yr <- add_edges(g_yr, edges_to_add, weight = 0)
    
    # Reken nu 1 keer de afstanden uit vanaf de SUPER_START. Dit elimineert de matrix-berekening!
    dist_vector <- distances(g_yr, v = "SUPER_START", mode = "all")[1, ]
    
    # Filter de nodes die binnen de afstand vallen (en gooi de SUPER_START er weer uit)
    valid_nodes <- names(dist_vector)[dist_vector <= max_dist_m & names(dist_vector) != "SUPER_START"]
    
    # F. Samenvoegen tot MULTILINESTRING
    message(paste(" -> Geometry samenvoegen (bereik:", length(valid_nodes), "nodes)..."))
    
    sub_g <- induced_subgraph(g_yr, valid_nodes)
    comps <- components(sub_g)
    
    node_clusters <- data.frame(
      node_id = names(comps$membership),
      cluster_id_temp = as.character(comps$membership)
    )
    
    cluster_segments <- net_sf %>%
      inner_join(node_clusters, by = c("from_node" = "node_id"))
    
    clusters_sf_yr <- cluster_segments %>%
      group_by(cluster_id_temp) %>%
      summarise(
        jaar = yr,
        n_segmenten = n(),
        totale_lengte_km = round(sum(weight) / 1000, 2),
        geometry = st_union(geometry),
        .groups = "drop"
      ) %>%
      mutate(cluster_id = paste0("Y", yr, "_C", cluster_id_temp)) %>%
      select(cluster_id, jaar, n_segmenten, totale_lengte_km, geometry) %>%
      st_cast("MULTILINESTRING")
    
    all_clusters[[as.character(yr)]] <- clusters_sf_yr
  }
  
  # --- 3. Resultaten bundelen ---
  if (length(all_clusters) > 0) {
    final_clusters_sf <- bind_rows(all_clusters)
    message(paste("\nKlaar! In totaal", nrow(final_clusters_sf), "clusters gegenereerd."))
    return(final_clusters_sf)
  } else {
    return(NULL)
  }
}

Heeft dit knelpunt een VHAS-code ingevuld?

Wat gebeurt er dus netto met een knelpunt?

Scenario B (Foute/Ontbrekende VHAS): Je knelpunt heeft geen VHAS (of een code die nergens in de rivier-shapefile voorkomt). Stap 2 kan niks vinden, dus hij laat het antwoord uit Stap 1 gewoon staan. Het knelpunt kleeft dan vast aan de absoluut dichtstbijzijnde waterloop, ongeacht de naam of code.

Door de trage fallback-berekening (die tegen 164.000 segmenten moet rekenen) in één keer (gevectoriseerd) uit te voeren, bespaar je R een loop waarin hij duizenden keren die zware berekening moet doen. Stap 2 is supersnel omdat hij daar nog maar hoeft te rekenen met het handjevol segmenten dat een specifieke VHAS-code deelt.
